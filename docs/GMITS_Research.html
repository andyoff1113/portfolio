GMITS: Grey Model for Irregular Time Series

Research Overview — Early-Stage Outbreak Forecasting

Andy Chu

2026-05-23 (last updated 2026-10-14)

GMITS extends the classical grey-system GM(1,1) model to irregularly spaced reporting intervals, aimed at early-stage outbreak forecasting when only a handful of data points are available. This page collects the shared methodology, checks it on synthetic data, and evaluates it on real-time snapshots of the 2022 England mpox outbreak. Earlier applications to two food-borne outbreaks are on the Bongkrekic Acid Food Poisoning (2024) and 2011 Listeria Outbreak pages.

# 2026-10-13 fix in gmits(): integrating dx1/dt + a * x1 = b over
# [t_{k-1}, t_k] gives x0_k * dt_k = -a * z_k * dt_k + b * dt_k, so the
# regression is x0_k = -a * z_k + b. The previous version used
# B <- cbind(-z * delta_t, delta_t), which scaled (a, b) by 1 / dt and
# was exact only for dt = 1 (it flipped the sign of a on some irregular
# grids). See D:/KnowledgeVault/research/grey_system/figures/gmits_dt_bug.png

mae <- function(actual, predicted) {
  mean(abs(actual - predicted), na.rm = TRUE)
}

mape <- function(actual, predicted) {
  valid <- actual != 0
  mean(
    abs((actual[valid] - predicted[valid]) / actual[valid]),
    na.rm = TRUE
  ) * 100
}

# n_ahead: out-of-sample steps using mean(delta_t) as step size.
# Returns $x0_future (and $t_future as numeric time offsets) when n_ahead > 0.
gmits <- function(count, date, n_ahead = 0) {
  valid <- !is.na(count) & !is.na(date)
  x0    <- count[valid]
  date  <- date[valid]
  if (length(x0) < 2) {
    warning("Not enough valid data points for modeling.")
    return(NULL)
  }
  if (x0[1] == 0) x0[1] <- 1e-3
  n <- length(x0)

  delta_t    <- as.numeric(
    difftime(date[2:n], date[1:(n - 1)], units = "days")
  )
  time_index <- cumsum(c(0, delta_t))

  x1 <- cumsum(c(x0[1], x0[2:n] * delta_t))
  z  <- 0.5 * (x1[2:n] + x1[1:(n - 1)])

  B <- cbind(-z, 1)
  Y <- x0[2:n]
  W <- diag(delta_t)
  p <- solve(t(B) %*% W %*% B) %*% t(B) %*% W %*% Y
  a_hat <- p[1]
  b_hat <- p[2]

  x1_hat <- c(
    x1[1],
    (x1[1] - b_hat / a_hat) *
      exp(-a_hat * (time_index[-1] - time_index[1])) +
      b_hat / a_hat
  )
  x0_hat <- c(x0[1], diff(x1_hat) / delta_t)

  result <- list(
    a = a_hat, b = b_hat,
    x1_hat = x1_hat, x0_hat = x0_hat,
    date = date, delta_t = delta_t
  )

  if (n_ahead > 0) {
    dt_mean  <- mean(delta_t)
    t_future <- time_index[n] + cumsum(rep(dt_mean, n_ahead))
    x1_fut   <- (x1[1] - b_hat / a_hat) *
      exp(-a_hat * (t_future - time_index[1])) + b_hat / a_hat
    result$x0_future   <- c(
      (x1_fut[1] - x1_hat[n]) / dt_mean,
      diff(x1_fut) / dt_mean
    )
    result$date_future <- date[n] + round(dt_mean * seq_len(n_ahead))
  }
  result
}

# n_ahead > 0: point/lower/upper each extend to n + n_ahead;
# last n_ahead entries are the out-of-sample prediction interval.
gmits_boot <- function(count, date, n_boot = 1000, ci = 0.95, n_ahead = 0) {
  base <- gmits(count, date, n_ahead = n_ahead)
  if (is.null(base)) return(NULL)

  valid      <- !is.na(count) & !is.na(date)
  x0         <- count[valid]
  dv         <- date[valid]
  n          <- length(x0)
  resid_pool <- (x0 - base$x0_hat)[2:n]
  n_out      <- n + n_ahead

  boot_mat <- replicate(n_boot, {
    e_boot  <- sample(resid_pool, n - 1, replace = TRUE)
    x0_boot <- c(x0[1], pmax(base$x0_hat[2:n] + e_boot, 0))
    res     <- tryCatch(
      gmits(x0_boot, dv, n_ahead = n_ahead),
      error = function(e) NULL
    )
    if (is.null(res)) return(rep(NA_real_, n_out))
    c(res$x0_hat, if (n_ahead > 0) res$x0_future else NULL)
  })

  alpha <- (1 - ci) / 2
  list(
    point = c(base$x0_hat,
              if (n_ahead > 0) base$x0_future else NULL),
    lower = apply(boot_mat, 1, quantile, probs = alpha,     na.rm = TRUE),
    upper = apply(boot_mat, 1, quantile, probs = 1 - alpha, na.rm = TRUE),
    a = base$a,
    b = base$b
  )
}

# Log-linear OLS baseline: log(y) ~ t.
# Only non-zero observations are used (consistent with gmits non-zero day approach).
# n_ahead: step forward using mean(delta_t) of the input dates.
exp_reg <- function(count, date, n_ahead = 0) {
  valid <- !is.na(count) & count > 0 & !is.na(date)
  x0 <- count[valid]
  dv <- date[valid]
  t_num <- as.numeric(dv - dv[1])
  fit <- lm(log(x0) ~ t_num)
  n <- length(x0)
  dt_mean <- if (n > 1) mean(diff(t_num)) else 1

  result <- list(
    alpha    = coef(fit)[["(Intercept)"]],
    beta     = coef(fit)[["t_num"]],
    x0_hat   = exp(fitted(fit)),
    date     = dv
  )
  if (n_ahead > 0) {
    t_future <- t_num[n] + cumsum(rep(dt_mean, n_ahead))
    result$x0_future   <- exp(
      predict(fit, newdata = data.frame(t_num = t_future))
    )
    result$date_future <- dv[n] + round(dt_mean * seq_len(n_ahead))
  }
  result
}

# Multiply the most recent n_recent observations by scale_factor
# before passing to gmits(), simulating upward correction of right-truncated data.
truncation_adjust <- function(count, n_recent = 1, scale_factor = 1.5) {
  adj <- count
  idx <- tail(seq_along(count), n_recent)
  adj[idx] <- count[idx] * scale_factor
  adj
}

Synthetic Validation

Validates that GMITS recovers known parameters from irregular-interval data with no noise. Two scenarios: growth phase (a < 0) and decay phase (a > 0).

# Generate x0 from ODE solution with known (a, b, x0[1]) and irregular spacing
synth_series <- function(a, b, x0_1, t_pts) {
  x1 <- (x0_1 - b / a) * exp(-a * t_pts) + b / a
  dt <- diff(t_pts)
  x0 <- c(x0_1, diff(x1) / dt)
  list(x0 = x0, dates = as.Date("2024-01-01") + t_pts)
}

# True ODE continuation for comparison in plot
true_oos <- function(a, b, x0_1, t_fit, n_ahead) {
  dt_mean  <- mean(diff(t_fit))
  t_fut    <- t_fit[length(t_fit)] + cumsum(rep(dt_mean, n_ahead))
  t_all    <- c(t_fit, t_fut)
  x1_all   <- (x0_1 - b / a) * exp(-a * t_all) + b / a
  dt_all   <- diff(t_all)
  c(x0_1, diff(x1_all) / dt_all)
}

Implementation note. An earlier version of this page attributed a much larger growth-phase underprediction to the background value; most of it was an implementation error, corrected on 2026-10-13 (revision history). With the corrected regression the growth scenario recovers \(\hat a =\) -0.357 (true \(-0.4\)) and the decay scenario \(\hat a =\) 0.293 (true \(0.3\)).

Remaining limitation — background-value bias in the growth phase. The residual gap in the growth scenario is the approximation bias of the GM(1,1) family. The background value \(z_{k} = \frac{1}{2}(x^{(1)}_{k} + x^{(1)}_{k-1})\) approximates the interval mean of \(x^{(1)}\) by the arithmetic mean of its endpoints. For an exponentially growing \(x^{(1)}\) (convex curve), the arithmetic mean of the endpoints understates the true integral mean, causing the model to estimate a slower growth rate than the truth. The bias scales with \(|a| \times \Delta t\): here \(|a| \times \overline{\Delta t} \approx 1\), which gives roughly a 10% underestimate of \(|a|\); at \(|a| \times \Delta t \approx 0.1\) it is below 1%. In the decay scenario \(x^{(1)}\) is concave, the bias acts in the opposite direction, and the absolute magnitude is smaller.

This limitation is well-known in the grey system literature (Liu, 2025; Liu & Deng, 2000) and is the motivation for optimising the background coefficient \(\alpha\) (replacing the fixed 0.5). That extension is outside the scope of this study; the bias is reported here as an explicit boundary condition for GMITS applicability: the method is most reliable when \(|a| \times \overline{\Delta t}\) is small, i.e., moderate growth rates and short reporting intervals.

Data: 2022 England mpox outbreak

Mpox spread from person to person, so early incidence grows roughly exponentially and the GMITS growth coefficient can be read as a growth rate. (The two food-borne outbreaks analysed earlier are common-source events whose epicurves are shaped by exposure timing and the incubation period, not by transmission; they are kept as supplementary examples on the Bongkrekic acid and Listeria pages.)

The data come from the repository accompanying Overton et al. (2023), Nowcasting the 2022 mpox outbreak in England (arXiv:2302.09076), released under CC0. Two parts are used:

  • Ground truth: daily cases by specimen date after all backfilling.
  • Real-time snapshots: the reporting triangle as it stood on day T, i.e. only the cases that had been reported by then.

Dates are anonymised in the source, so the x-axis is a day index. Symptom-onset snapshots start after the peak, so the specimen-date series is used. Daily counts drop sharply at weekends, so counts are summed into non-overlapping 7-day weeks ending on the snapshot day.

Early-Stage Forecasting

At each snapshot the last 5 weeks are used as input (n = 5 points, Δt = 7 days) and the next 2 weeks are forecast. Three inputs are compared:

  • Reported: counts as available on day T (right-truncated).
  • Reported, last week ×1.5: a simple truncation adjustment.
  • Oracle: the backfilled counts for the same weeks, i.e. what a perfect truncation correction would give.

The exponential regression baseline is fitted to the reported input.

n_fit_m <- 5
n_ahead_m <- 2

run_snapshot <- function(t_snap) {
  fit_ends <- t_snap - 7 * rev(seq_len(n_fit_m) - 1)
  fut_ends <- t_snap + 7 * seq_len(n_ahead_m)
  fit_dates <- as.Date("2022-01-01") + fit_ends

  obs <- week_sum(snapshot_daily(t_snap), fit_ends)
  inputs <- list(
    "GMITS (reported)" = obs,
    "GMITS (reported, ×1.5)" = truncation_adjust(obs, 1, 1.5),
    "GMITS (oracle)" = week_sum(truth_daily, fit_ends)
  )

  fc <- imap(inputs, function(x, nm) {
    res <- gmits_boot(x, fit_dates, n_boot = 1000, n_ahead = n_ahead_m)
    tibble(
      method = nm,
      input = c(x, rep(NA, n_ahead_m)),
      point = res$point,
      lower = res$lower,
      upper = res$upper
    )
  })

  # exp_reg() drops zero weeks, so rebuild fitted values on the full grid
  er <- exp_reg(obs, fit_dates)
  t_all <- as.numeric(
    as.Date("2022-01-01") + c(fit_ends, fut_ends) - er$date[1]
  )
  fc$exp_reg <- tibble(
    method = "Exp. regression (reported)",
    input = c(obs, rep(NA, n_ahead_m)),
    point = exp(er$alpha + er$beta * t_all),
    lower = NA_real_,
    upper = NA_real_
  )

  bind_rows(fc) |>
    group_by(method) |>
    mutate(
      step = seq_len(n()) - n_fit_m,
      week_end = c(fit_ends, fut_ends),
      truth = week_sum(truth_daily, c(fit_ends, fut_ends))
    ) |>
    ungroup() |>
    mutate(snapshot = t_snap)
}

Three points stand out. First, with complete (oracle) input, GMITS forecasts one week ahead with a MAPE under 20% from only five weekly points. Second, on the input actually available in real time it does worse than the exponential regression, and right truncation, not the model form, is the dominant source of error. Third, with five points the bootstrap intervals are unreliable: coverage is well below the nominal 95%, so the intervals should not be read as calibrated until the simulation study planned for this project has quantified their coverage.

2026-10-13 Corrected the gmits() regression. Integrating \(dx^{(1)}/dt + a x^{(1)} = b\) over each interval gives \(x^{(0)}_{k} = -a z_{k} + b\); the previous code regressed on \((-z_{k}\Delta t_{k}, \Delta t_{k})\), which shrank \((\hat a, \hat b)\) by \(1/\Delta t\) whenever reporting intervals were not one day (and could flip the sign of \(\hat a\) on irregular grids). All GMITS pages now load one shared function file. Synthetic growth scenario: \(\hat a\) −0.116 → −0.357 (true −0.4). Listeria (n = 6) MAPE 90.25% → 45.45% (exp. regression 50.20%). Bongkrek (n = 4) MAPE 81.19% → 58.05% (exp. regression 64.47%). With 4 and 1 verifiable steps, these do not establish that either method is more accurate.
